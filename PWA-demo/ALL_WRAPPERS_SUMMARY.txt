    console.log('[SL] Using Electron proxy');

    console.log('[SL] Using Tauri proxy');

  }
  
  // 4. Browser - Direct (CORS blocked)
  console.log('[SL] Direct fetch - requires external CORS proxy');
  return await fetch(url, {...});
}
